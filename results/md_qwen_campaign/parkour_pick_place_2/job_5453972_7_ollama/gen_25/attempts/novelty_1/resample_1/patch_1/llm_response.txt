<DESCRIPTION>
This approach simplifies the task by breaking it into clear steps. It ensures that every contact transition is valid by explicitly managing the contact states. This approach avoids the common pitfalls of the previous implementations by ensuring that the robot's limbs are properly released before transitioning to a new contact. Additionally, it uses a more structured approach to define the steps of the task, making it easier to reason about the sequence of actions.

from __future__ import annotations

from typing import Any, Dict, List, Tuple

def generate_contact_plan(scene: List[Dict[str, Any]]) -> Dict[str, List[Tuple]]:
    plan = get_initial_mode()

    # Walk to the white box
    walk(plan, 5)

    # Reach for the white box with the right hand
    append_mode(plan, right_ee=(14, 1))

    # Pick up the white box
    append_mode(plan, right_ee=(14, 1), white_box=(1, 14, 1))

    # Walk to the elevated table
    walk(plan, 5)

    # Place the white box on the elevated table
    append_mode(plan, right_ee=(), white_box=(6, 8, 1))

    return plan

# EVOLVE-BLOCK-END

from env import append_mode, get_default_scene, get_initial_mode, walk

def run_planning():
    scene = get_default_scene()
    contact_plan = generate_contact_plan(scene)
    return contact_plan, scene
