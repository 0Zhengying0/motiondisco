A modularized approach to generate a contact plan for a humanoid robot, focusing on clear separation of tasks and improved readability and maintainability.

<DESCRIPTION>
The current implementation is monolithic and lacks clarity in its task separation, making it difficult to maintain and extend. This redesign introduces a modular architecture where each phase of the task (picking, walking, and placing) is encapsulated within its own function. This separation not only enhances readability but also facilitates easier debugging and potential future enhancements. Additionally, the use of a `Task` class helps in encapsulating the state and methods related to the plan generation, ensuring that the plan is built step-by-step in a controlled manner.
</DESCRIPTION>

from __future__ import annotations

from typing import Any, Dict, List, Tuple

class Task:
    def __init__(self):
        self.plan = get_initial_mode()

    def pick_box(self):
        # Approach the box
        self.append_mode(left_ee=(14, 1), right_ee=(14, 1))
        # Lift the box
        self.append_mode(left_ee=(), right_ee=(), left_sole=(1, 1), right_sole=(1, 1))
        # Move the box to the front of the robot
        self.append_mode(left_ee=(14, 1), right_ee=(), left_sole=(1, 1), right_sole=(1, 1))

    def walk_to_table(self):
        # Walk to the table
        walk(self.plan, 4)

    def place_box(self):
        # Move the box towards the table
        self.append_mode(left_ee=(14, 1), right_ee=(), left_sole=(1, 1), right_sole=(1, 1))
        # Place the box on the table
        self.append_mode(left_ee=(), right_ee=(), left_sole=(8, 1), right_sole=(8, 1), white_box=(6, 8, 1))

    def append_mode(self, **channel_updates) -> None:
        append_mode(self.plan, **channel_updates)

    def generate_plan(self) -> Dict[str, List[Tuple]]:
        self.pick_box()
        self.walk_to_table()
        self.place_box()
        return self.plan

def generate_contact_plan(scene: List[Dict[str, Any]]) -> Dict[str, List[Tuple]]:
    task = Task()
    return task.generate_plan()
