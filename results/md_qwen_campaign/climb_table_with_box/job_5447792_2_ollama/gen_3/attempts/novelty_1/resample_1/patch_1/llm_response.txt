def generate_contact_plan(scene: List[Dict[str, Any]]) -> Dict[str, List[Tuple]]:
    """
    Build a contact_mode_seq for the active task — see the system
    prompt for scene-specific goals.

    The body below returns the initial mode unchanged (1 mode, which
    scores 0 since a plan needs at least 2 modes); evolution is
    expected to extend it into a full plan.
    """
    plan = get_initial_mode()

    # Step 1: Lift the left foot off the floor
    append_mode(plan, left_sole=())

    # Step 2: Move the left foot towards the table
    append_mode(plan, left_sole=(2, 1))

    # Step 3: Lift the right foot off the floor
    append_mode(plan, right_sole=())

    # Step 4: Move the right foot towards the table
    append_mode(plan, right_sole=(2, 1))

    # Step 5: Lift the left hand off the box
    append_mode(plan, left_ee=())

    # Step 6: Move the left hand towards the box
    append_mode(plan, left_ee=(9, 1))

    # Step 7: Lift the right hand off the box
    append_mode(plan, right_ee=())

    # Step 8: Move the right hand towards the box
    append_mode(plan, right_ee=(11, 1))

    return plan
