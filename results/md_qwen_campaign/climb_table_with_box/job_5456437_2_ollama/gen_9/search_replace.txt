def generate_contact_plan(scene: List[Dict[str, Any]]) -> Dict[str, List[Tuple]]:
    """
    Build a contact_mode_seq for the active task — see the system
    prompt for scene-specific goals.

    The body below returns the initial mode unchanged (1 mode, which
    scores 0 since a plan needs at least 2 modes); evolution is
    expected to extend it into a full plan.
    """
    plan = get_initial_mode()

    # Move the robot closer to the table
    walk(plan, 5)

    # Lift the robot's base to prepare for climbing
    append_mode(plan, left_sole=(1, 2), right_sole=(1, 2))

    # Move the robot forward onto the table
    walk(plan, 2)

    # Release the box and grab it with the left hand
    append_mode(plan, left_ee=(9, 1), right_ee=(), white_box=())

    # Move the robot backward to balance
    walk(plan, 2)

    # Move the robot forward to the final position
    walk(plan, 2)

    return plan
