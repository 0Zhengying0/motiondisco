def ensure_final_mode(plan: Dict[str, List[Tuple]]) -> None:
    """Ensure the final mode meets the discrete contact goal."""
    final_mode = {
        "left_sole": [(1, 1)],
        "right_sole": [(1, 1)],
    }
    for channel, contacts in final_mode.items():
        plan[channel].extend([contacts[-1]] * (len(plan[channel]) - len(contacts)))
